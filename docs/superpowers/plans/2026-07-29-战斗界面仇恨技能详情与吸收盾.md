# 战斗界面仇恨、技能详情与吸收盾实施计划

> **供执行代理使用：** 按任务顺序执行，并使用复选框（`- [ ]`）跟踪进度。

**目标：** 通过向内绘制的仇恨边框、结构化技能基础数值、实时吸收盾条和 RD-4 教程补充，提高战斗信息可读性。

**架构：** 视觉状态保持在现有 React UI 组件内。新增纯函数 `skillDetailFacts` 格式化器，通过显式 `skillLogicId` 注册表把结构化主动技能定义和启用效果行转换为可渲染事实，不解析描述文本、不修改策划数据。总吸收盾直接从当前玩家状态计算，敌人边框仅用 CSS 向内绘制。

**技术栈：** React 19、TypeScript 6、Vitest、JSDOM、CSS、Vite。

**Workspace note:** Execute in the current dirty `main` because current Bear T runtime behavior depends on uncommitted planner-owned workbook changes. Do not modify or regenerate `public/`; do not commit implementation files unless the user requests it.

```ts
import { describe, expect, it } from 'vitest'
import type { ActiveSkillDefinition, ActiveSkillEffectDefinition } from '../game/encounter/encounterTypes'
import { buildSkillDetailFacts } from './skillDetailFacts'

const skill = (overrides: Partial<ActiveSkillDefinition>): ActiveSkillDefinition => ({
  id: 'test_skill', name: '测试技能', shortName: '测试', description: '', iconId: 'test',
  cooldownMs: 9000, initialRemainingCooldownMs: 0, resourceCost: 0, gcdMs: 1500,
  pointCost: 4, targetingType: 'enemy', skillLogicId: 'bear_thrash',
  castStopMode: 'none', canAffectSkull: false, grantedStatusIds: [], ...overrides,
})

const effect = (overrides: Partial<ActiveSkillEffectDefinition>): ActiveSkillEffectDefinition => ({
  skillEffectId: 'effect', skillId: 'test_skill', effectIndex: 1, targetSelector: 'cross',
  enabled: true, ...overrides,
})

describe('buildSkillDetailFacts', () => {
  it('formats cooldown, Bear damage, threat, and rage from base definitions', () => {
    expect(buildSkillDetailFacts(skill({}), [effect({ valueA: 10, valueB: 10, threatMultiplier: 5 })]))
      .toEqual(expect.arrayContaining([
        { key: 'cooldown', label: '技能冷却', value: '9 秒' },
        { key: 'damage', label: '技能伤害', value: '10' },
        { key: 'threatMultiplier', label: '仇恨倍数', value: '×5' },
        { key: 'resourceGain', label: '怒气获取', value: '10' },
      ]))
  })

  it('formats mitigation, healing, shield, duration, and Warrior primary/secondary damage', () => {
    expect(buildSkillDetailFacts(
      skill({ skillLogicId: 'bear_barkskin' }),
      [effect({ valueB: 0.3, durationMs: 10_000 })],
    )).toEqual(expect.arrayContaining([
      { key: 'damageReduction', label: '全伤害减免', value: '30%' },
      { key: 'duration', label: '持续时间', value: '10 秒' },
    ]))

    expect(buildSkillDetailFacts(
      skill({ skillLogicId: 'bear_regrowth' }),
      [
        effect({ effectIndex: 1, valueA: 25, valueB: 5, durationMs: 6000 }),
        effect({ skillEffectId: 'tick', effectIndex: 2, valueA: 5, valueB: 3, durationMs: 6000 }),
      ],
    )).toEqual(expect.arrayContaining([
      { key: 'healing', label: '即时治疗', value: '25' },
      { key: 'healingPerTick', label: '每跳治疗', value: '5' },
      { key: 'healingTicks', label: '治疗跳数', value: '3' },
    ]))

    expect(buildSkillDetailFacts(
      skill({ skillLogicId: 'ignore_pain' }),
      [effect({ valueA: 25, valueB: 0.5, durationMs: 5000 })],
    )).toEqual(expect.arrayContaining([
      { key: 'absorb', label: '吸收盾', value: '25' },
      { key: 'absorbRatio', label: '伤害吸收比例', value: '50%' },
    ]))
  })

  it('omits zero cooldown, disabled effects, duplicate threat multipliers, and unknown value semantics', () => {
    const facts = buildSkillDetailFacts(
      skill({ cooldownMs: 0, skillLogicId: 'unknown_logic' }),
      [
        effect({ valueA: 999, valueB: 888, threatMultiplier: 5 }),
        effect({ skillEffectId: 'duplicate', effectIndex: 2, threatMultiplier: 5 }),
        effect({ skillEffectId: 'disabled', effectIndex: 3, threatMultiplier: 9, enabled: false }),
      ],
    )

    expect(facts).toEqual([
      { key: 'threatMultiplier', label: '仇恨倍数', value: '×5' },
    ])
  })
})
```
